JavaScript Essentials — Quick Cheatsheet

Variables & Data Types

Declaration

let name = "John";
const age = 25;
var oldWay = "avoid this";

Data Types

  • string - "hello"
  • number - 42, 3.14
  • boolean - true, false
  • undefined - not assigned
  • null - intentionally empty
  • object - {key: "value"}
  • array - [1, 2, 3]

Functions

Function Declaration

function greet(name) {
  return `Hello, ${name}!`;
}

Arrow Function

const greet = (name) => {
  return `Hello, ${name}!`;
};

// Shorthand
const greet = name => `Hello, ${name}!`;

Function Expression

const greet = function(name) {
  return `Hello, ${name}!`;
};

Arrays

Common Methods

const arr = [1, 2, 3];

// Add/Remove
arr.push(4);        // [1, 2, 3, 4]
arr.pop();          // [1, 2, 3]
arr.unshift(0);     // [0, 1, 2, 3]
arr.shift();        // [1, 2, 3]

// Transform
arr.map(x => x * 2);     // [2, 4, 6]
arr.filter(x => x > 1);  // [2, 3]
arr.reduce((a, b) => a + b); // 6

Objects

Object Literal

const person = {
  name: "John",
  age: 25,
  greet() {
    return `Hi, I'm ${this.name}`;
  }
};

Accessing Properties

person.name;           // "John"
person["age"];        // 25
const key = "name";
person[key];          // "John"

Control Flow

If/Else

if (age >= 18) {
  console.log("Adult");
} else if (age >= 13) {
  console.log("Teen");
} else {
  console.log("Child");
}

Ternary Operator

const status = age >= 18 ? "Adult" : "Minor";

Switch

switch (day) {
  case "Monday":
    console.log("Start of week");
    break;
  case "Friday":
    console.log("TGIF!");
    break;
  default:
    console.log("Regular day");
}

Loops

For Loop

for (let i = 0; i < 5; i++) {
  console.log(i);
}

For...of (Arrays)

for (const item of arr) {
  console.log(item);
}

For...in (Objects)

for (const key in person) {
  console.log(key, person[key]);
}

While Loop

let i = 0;
while (i < 5) {
  console.log(i);
  i++;
}

DOM Manipulation

Selecting Elements

document.getElementById("myId");
document.querySelector(".myClass");
document.querySelectorAll("p");

Modifying Content

element.textContent = "New text";
element.innerHTML = "Bold";
element.style.color = "red";

Event Listeners

button.addEventListener("click", () => {
  console.log("Button clicked!");
});

Async/Await

Promise

fetch("https://api.example.com/data")
  .then(response => response.json())
  .then(data => console.log(data))
  .catch(error => console.error(error));

Async/Await

async function fetchData() {
  try {
    const response = await fetch("https://api.example.com/data");
    const data = await response.json();
    console.log(data);
  } catch (error) {
    console.error(error);
  }
}

Destructuring

Array Destructuring

const [first, second, ...rest] = [1, 2, 3, 4, 5];
// first = 1, second = 2, rest = [3, 4, 5]

Object Destructuring

const {name, age, ...other} = person;
// name = person.name, age = person.age

💡 Pro Tips

  • Use const by default, let when reassigning
  • Prefer arrow functions for short, simple functions
  • Use template literals with backticks for string interpolation
  • Always handle errors in async operations
  • Use === instead of == for strict equality
